import os
import time
import hmac

import logging
from datetime import datetime, timezone, timedelta
import requests

# ============================================================

TELEGRAM_TOKEN = os.getenv("TELEGRAM_TOKEN", "").strip()
TELEGRAM_CHAT_ID = os.getenv("TELEGRAM_CHAT_ID", "").strip()
XT_API_KEY = os.getenv("XT_API_KEY", "").strip()
XT_API_SECRET = os.getenv("XT_API_SECRET", "").strip()
XT_BASE_URL = "https://fapi.xt.com"

CHECK_INTERVAL = 120
MAX_AGE_SECONDS = 120
IRAN_TZ = timezone(timedelta(hours=3, minutes=30))

seen_positions = set()

logging.basicConfig(level=logging.INFO, format="%(asctime)s | %(levelname)s | %(message)s")
log = logging.getLogger("REZA-MONITOR")

def canonical_form(params):
    if not params:
        return ""
    pairs = []
    for key in sorted(params.keys()):
        value = params[key]
        if value is None:
            continue
        if isinstance(value, bool):
            value = "true" if value else "false"
        pairs.append(f"{key}={value}")
    return "&".join(pairs)

def calculate_signature(api_key, api_secret, timestamp, path, query_string="", body_string=""):
    x = "xt-validate-appkey=" + str(api_key) + "&xt-validate-timestamp=" + str(timestamp)
    parts = [path]
    if query_string:
        parts.append(query_string)
    if body_string:
        parts.append(body_string)
    y = "#".join(parts)
    original = x + "#" + y
    return hmac.new(api_secret.encode("utf-8"), original.encode("utf-8"), hashlib.sha256).hexdigest()

def xt_private_get(path, params=None):
    params = params or {}
    timestamp = str(int(time.time() * 1000))
    query_string = canonical_form(params)
    signature = calculate_signature(XT_API_KEY, XT_API_SECRET, timestamp, path, query_string=query_string, body_string="")
    headers = {
        "Content-Type": "application/x-www-form-urlencoded",
        "xt-validate-appkey": XT_API_KEY,
        "xt-validate-timestamp": timestamp,
        "xt-validate-signature": signature,
        "xt-validate-recvwindow": "5000",
        "xt-validate-algorithms": "HmacSHA256",
        "xt-validate-signversion": "1.0"
    }
    response = requests.get(XT_BASE_URL + path, params=params, headers=headers, timeout=10)
    if response.status_code >= 400:
        raise RuntimeError(f"XT HTTP {response.status_code}: {response.text[:500]}")
    data = response.json()
    rc = data.get("returnCode")
    if rc not in (None, 0, "0"):
        raise RuntimeError(f"XT API error: {data}")
    return data

def get_xt_positions():
    try:
        response = xt_private_get("/future/user/v1/position/list", {})
        result = response.get("result", [])
        if isinstance(result, list):
            return result
        if isinstance(result, dict):
            return result.get("list", [])
        return []
    except Exception as exc:
        log.error(f"XT position error: {exc}")
        return []

def get_conditional_orders():
    """دریافت سفارشات شرطی (استاپ و تی‌پی)"""
    try:
        response = xt_private_get("/future/trade/v1/entrust/plan-list", {"state": "NOT_TRIGGERED"})
        result = response.get("result", {})
        if isinstance(result, dict):
            return result.get("items", [])
        if isinstance(result, list):
            return result
        return []
    except Exception as exc:
        log.error(f"XT conditional orders error: {exc}")
        return []

def get_open_orders():
    """دریافت سفارشات باز معمولی"""
    try:
        response = xt_private_get("/future/trade/v1/order/list-open-order", {})
        result = response.get("result", [])
        if isinstance(result, list):
            return result
        if isinstance(result, dict):
            return result.get("list", [])
        return []
    except Exception as exc:
        log.error(f"XT open orders error: {exc}")
        return []

# پیدا کردن استاپ و تی‌پی
# ============================================================

def find_stop_tp(symbol, side, conditional_orders, open_orders):
    """
    پیدا کردن استاپ و تی‌پی برای یه پوزیشن
    - استاپ از سفارشات شرطی (STOP_MARKET)
    - تی‌پی از سفارشات باز (LIMIT)
    """
    stop = None
    tp = None
    symbol_lower = symbol.lower()
    side_upper = side.upper()
    
    # برای LONG: بستن با SELL
    # برای SHORT: بستن با BUY
    close_side = "SELL" if side_upper == "LONG" else "BUY"
    
    # 1. استاپ از سفارشات شرطی
    for order in conditional_orders:
        if not isinstance(order, dict):
            continue
        if str(order.get("symbol", "")).lower() != symbol_lower:
            continue
        if str(order.get("orderSide", "")).upper() != close_side:
            continue
        
        entrust_type = str(order.get("entrustType", "")).upper()
        if "STOP" in entrust_type:
            price = order.get("stopPrice")
            if price:
                stop = float(price)
                break
    
    # 2. تی‌پی از سفارشات باز (LIMIT)
    for order in open_orders:
        if not isinstance(order, dict):
            continue
        if str(order.get("symbol", "")).lower() != symbol_lower:
            continue
        if str(order.get("orderSide", "")).upper() != close_side:
            continue
        
        order_type = str(order.get("orderType", "")).upper()
        if "LIMIT" in order_type:
            price = order.get("price")
            if price:
                tp = float(price)
                break
    
    return stop, tp

def send_telegram(message):
    if not TELEGRAM_TOKEN or not TELEGRAM_CHAT_ID:
        return False
    try:
        url = f"https://api.telegram.org/bot{TELEGRAM_TOKEN}/sendMessage"
        r = requests.post(url, json={"chat_id": TELEGRAM_CHAT_ID, "text": message, "parse_mode": "HTML"}, timeout=5)
        if r.status_code == 200:
            log.info("✅ Telegram sent")
            return True
        else:
            log.error(f"❌ Telegram error: {r.text}")
            return False
    except Exception as e:
        log.error(f"❌ Telegram exception: {e}")
        return False

def format_position_message(pos, stop, tp):
    side = pos.get("positionSide", "UNKNOWN")
    entry = float(pos.get("entryPrice", 0) or 0)
    lev = int(pos.get("leverage", 0) or 0)
    symbol = pos.get("symbol", "UNKNOWN").upper().replace("_", "")
    
    stop_str = f"{stop:.2f}" if stop else "—"
    tp_str = f"{tp:.2f}" if tp else "—"
    
    side_emoji = "🟢" if side == "LONG" else "🔴"
    direction_text = "لانگ (خرید)" if side == "LONG" else "شورت (فروش)"
    
    now_iran = datetime.now(IRAN_TZ)
    time_str = now_iran.strftime('%Y-%m-%d %H:%M:%S')
    line = "━━━━━━━━━━━━━━━━━━━━━━━━"
    
    return f"""🆕 <b>پوزیشن جدید</b>

{line}
{side_emoji} <b>{symbol}</b>  |  <b>{direction_text}</b>
{line}

💰 <b>قیمت ورود:</b>  <code>{entry:.2f}</code> USDT
🛑 <b>حد ضرر:</b>  <code>{stop_str}</code> USDT
🎯 <b>حد سود:</b>  <code>{tp_str}</code> USDT
📦 <b>حجم:</b>  <i>دلخواه با مدیریت سرمایه</i>
⚡ <b>اهرم:</b>  <code>{lev}x</code>
📈 <b>سود/زیان:</b>  <b>1.5 سود / 1 ضرر به نسبت</b>

{line}
🕐 <code>{time_str}</code>
📌 <i>مدیریت سرمایه و پوزیشن با شماست</i>"""

def check_once():
    global seen_positions
    try:
        now = time.time()
        positions = get_xt_positions()
        conditional_orders = get_conditional_orders()
        open_orders = get_open_orders()
        current_keys = set()
        
        for pos in positions:
            size = float(pos.get("positionSize", 0) or 0)
            if size <= 0:
                continue
            
            symbol = pos.get("symbol", "UNKNOWN").upper().replace("_", "")
            side = pos.get("positionSide", "UNKNOWN")
            entry = float(pos.get("entryPrice", 0) or 0)
            key = f"{symbol}_{side}"
            current_keys.add(key)
            
            updated_ms = pos.get("updatedTime")
            age_seconds = None
            if updated_ms:
                try:
                    ts = int(updated_ms)
                    if ts > 1e12:
                        ts = ts / 1000
                    age_seconds = now - ts
                except:
                    pass
            
            if key not in seen_positions:
                if age_seconds is None or age_seconds <= MAX_AGE_SECONDS:
                    stop, tp = find_stop_tp(symbol, side, conditional_orders, open_orders)
                    log.info(f"🆕 {key} | Entry: {entry} | Stop: {stop} | TP: {tp}")
                    send_telegram(format_position_message(pos, stop, tp))
                    seen_positions.add(key)
                    time.sleep(0.5)
                else:
                    log.info(f"⏸️ Old: {key} ({age_seconds:.0f}s)")
                    seen_positions.add(key)
            else:
                log.info(f"⏸️ Already sent: {key}")
        
        closed = seen_positions - current_keys
        if closed:
            log.info(f"🧹 Closed: {closed}")
            seen_positions -= closed
        
        log.info(f"📊 Active: {len(current_keys)} | Memory: {len(seen_positions)}")
        
    except Exception as exc:
        log.error(f"❌ Error: {exc}")

# ============================================================
# حلقه اصلی
# ============================================================

def main():
    log.info("🚀 Starting monitor...")
    while True:
        try:
            check_once()
        except Exception as exc:
            log.error(f"❌ Loop error: {exc}")
        log.info(f"⏳ Sleeping {CHECK_INTERVAL}s...")
        time.sleep(CHECK_INTERVAL)

if __name__ == "__main__":
    try:
        main()
    except KeyboardInterrupt:
        print("\n🛑 Stopped.")
    except Exception as exc:
        print(f"❌ Fatal: {exc}")
