import os
import time
import hmac
import hashlib
import logging
from datetime import datetime, timezone, timedelta
import requests

# ============================================================
# تنظیمات
# ============================================================

TELEGRAM_TOKEN = os.getenv("TELEGRAM_TOKEN", "").strip()
TELEGRAM_CHAT_ID = os.getenv("TELEGRAM_CHAT_ID", "").strip()
XT_API_KEY = os.getenv("XT_API_KEY", "").strip()
XT_API_SECRET = os.getenv("XT_API_SECRET", "").strip()
XT_BASE_URL = "https://fapi.xt.com"

CHECK_INTERVAL = 30
MAX_POSITION_AGE = 90
IRAN_TZ = timezone(timedelta(hours=3, minutes=30))

# ============================================================
# 🎯 استاپ سفارشی برای هر سمبل
# ============================================================

STOP_OFFSETS = {
    "ETHUSDT": 12,
    "BNBUSDT": 12,
    "BTCUSDT": 450,
    "HYPEUSDT": 0.7,
    "LINKUSDT": 0.22,
    "SOLUSDT": 1.5,
    "XRPUSDT": 0.04,
    "ZECUSDT": 15,
}

# 🧠 حافظه
START_TIME = time.time()
seen_positions = set()

logging.basicConfig(level=logging.INFO, format="%(asctime)s | %(levelname)s | %(message)s")
log = logging.getLogger("REZA-MONITOR")

# ============================================================
# توابع XT
# ============================================================

def canonical_form(params):
    if not params:
        return ""
    pairs = []
    for key in sorted(params.keys()):
        value = params[key]
        if value is None:
            continue
        if isinstance(value, bool):
            value = "true" if value else "false"
        pairs.append(f"{key}={value}")
    return "&".join(pairs)

def calculate_signature(api_key, api_secret, timestamp, path, query_string="", body_string=""):
    x = "xt-validate-appkey=" + str(api_key) + "&xt-validate-timestamp=" + str(timestamp)
    parts = [path]
    if query_string:
        parts.append(query_string)
    if body_string:
        parts.append(body_string)
    y = "#".join(parts)
    original = x + "#" + y
    return hmac.new(api_secret.encode("utf-8"), original.encode("utf-8"), hashlib.sha256).hexdigest()

def xt_private_get(path, params=None):
    params = params or {}
    timestamp = str(int(time.time() * 1000))
    query_string = canonical_form(params)
    signature = calculate_signature(XT_API_KEY, XT_API_SECRET, timestamp, path, query_string=query_string, body_string="")
    headers = {
        "Content-Type": "application/x-www-form-urlencoded",
        "xt-validate-appkey": XT_API_KEY,
        "xt-validate-timestamp": timestamp,
        "xt-validate-signature": signature,
        "xt-validate-recvwindow": "5000",
        "xt-validate-algorithms": "HmacSHA256",
        "xt-validate-signversion": "1.0"
    }
    response = requests.get(XT_BASE_URL + path, params=params, headers=headers, timeout=10)
    if response.status_code >= 400:
        raise RuntimeError(f"XT HTTP {response.status_code}: {response.text[:500]}")
    data = response.json()
    rc = data.get("returnCode")
    if rc not in (None, 0, "0"):
        raise RuntimeError(f"XT API error: {data}")
    return data

def get_xt_positions():
    try:
        response = xt_private_get("/future/user/v1/position/list", {})
        result = response.get("result", [])
        if isinstance(result, list):
            return result
        if isinstance(result, dict):
            return result.get("list", [])
        return []
    except Exception as exc:
        log.error(f"XT position error: {exc}")
        return []

def get_recent_orders():
    try:
        response = xt_private_get("/future/trade/v1/order/list-history", {
            "direction": "NEXT", "limit": 100
        })
        result = response.get("result", {})
        if isinstance(result, dict):
            return result.get("items", [])
        return []
    except Exception as exc:
        log.error(f"XT history error: {exc}")
        return []

# ============================================================
# زمان باز شدن پوزیشن
# ============================================================

def get_position_open_time(symbol, side, history_orders):
    symbol_lower = symbol.lower()
    side_upper = side.upper()
    entry_side = "BUY" if side_upper == "LONG" else "SELL"
    
    open_time = None
    for order in history_orders:
        if not isinstance(order, dict):
            continue
        if str(order.get("symbol", "")).lower() != symbol_lower:
            continue
        if str(order.get("orderSide", "")).upper() != entry_side:
            continue
        if str(order.get("positionSide", "")).upper() != side_upper:
            continue
        if str(order.get("state", "")).upper() not in ("FILLED", "PARTIALLY_FILLED"):
            continue
        if float(order.get("executedQty", 0) or 0) <= 0:
            continue
        
        created_ms = order.get("createdTime")
        if created_ms:
            try:
                ts = int(created_ms)
                if ts > 1e12:
                    ts = ts / 1000
                if open_time is None or ts > open_time:
                    open_time = ts
            except:
                continue
    return open_time

# ============================================================
# محاسبه استاپ سفارشی
# ============================================================

def calculate_stop(symbol, side, entry):
    """
    - LONG: entry - offset
    - SHORT: entry + offset
    """
    offset = STOP_OFFSETS.get(symbol.upper())
    if offset is None:
        return None
    if side.upper() == "LONG":
        return entry - offset
    else:
        return entry + offset

def send_telegram(message):
    if not TELEGRAM_TOKEN or not TELEGRAM_CHAT_ID:
        return False
    try:
        url = f"https://api.telegram.org/bot{TELEGRAM_TOKEN}/sendMessage"
        r = requests.post(url, json={"chat_id": TELEGRAM_CHAT_ID, "text": message, "parse_mode": "HTML"}, timeout=5)
        return r.status_code == 200
    except:
        return False

def format_position_message(pos):
    side = pos.get("positionSide", "UNKNOWN")
    entry = float(pos.get("entryPrice", 0) or 0)
    lev = int(pos.get("leverage", 0) or 0)
    symbol = pos.get("symbol", "UNKNOWN").upper().replace("_", "")
    
    stop = calculate_stop(symbol, side, entry)
    stop_str = f"{stop:.2f}" if stop else "—"
    
    side_emoji = "🟢" if side == "LONG" else "🔴"
    direction_text = "لانگ (خرید)" if side == "LONG" else "شورت (فروش)"
    
    now_iran = datetime.now(IRAN_TZ)
    time_str = now_iran.strftime('%Y-%m-%d %H:%M:%S')
    line = "━━━━━━━━━━━━━━━━━━━━━━━━"
    
    return f"""🆕 <b>پوزیشن جدید</b>

{line}
{side_emoji} <b>{symbol}</b>  |  <b>{direction_text}</b>
{line}

💰 <b>قیمت ورود:</b>  <code>{entry:.2f}</code> USDT
🛑 <b>حد ضرر:</b>  <code>{stop_str}</code> USDT
🎯 <b>حد سود:</b>  <b>Scalp</b>
📦 <b>حجم:</b>  <i>دلخواه با مدیریت سرمایه</i>
⚡ <b>اهرم:</b>  <code>{lev}x</code>

{line}
⚠️ <i>پوزیشن اسکلپ را در سود مناسب ببندید احتمال برگشت قیمت وجود دارد</i>
{line}
🕐 <code>{time_str}</code>
📌 <i>مدیریت سرمایه و پوزیشن با شماست</i>"""

# یک بار چک
# ============================================================

def check_once():
    global seen_positions
    try:
        now = time.time()
        positions = get_xt_positions()
        history_orders = get_recent_orders()
        current_keys = set()
        
        for pos in positions:
            size = float(pos.get("positionSize", 0) or 0)
            if size <= 0:
                continue
            
            symbol = pos.get("symbol", "UNKNOWN").upper().replace("_", "")
            side = pos.get("positionSide", "UNKNOWN")
            key = f"{symbol}_{side}"
            current_keys.add(key)
            
            if key in seen_positions:
                continue
            
            open_time = get_position_open_time(symbol, side, history_orders)
            if open_time is None:
                seen_positions.add(key)
                continue
            
            position_age = now - open_time
            if position_age > MAX_POSITION_AGE or open_time < START_TIME:
                seen_positions.add(key)
                continue
            
            log.info(f"🆕 Sending: {key} | Age: {position_age:.0f}s")
            send_telegram(format_position_message(pos))
            seen_positions.add(key)
            time.sleep(0.3)
        
        seen_positions -= (seen_positions - current_keys)
        
        log.info(f"📊 Active: {len(current_keys)} | Sent: {len(seen_positions)}")
        
    except Exception as exc:
        log.error(f"❌ Error: {exc}")

# ============================================================
# حلقه اصلی
# ============================================================

def main():
    log.info(f"🚀 Starting monitor | START_TIME={START_TIME}")
    while True:
        try:
            check_once()
        except Exception as exc:
            log.error(f"❌ Loop error: {exc}")
        time.sleep(CHECK_INTERVAL)

if __name__ == "__main__":
    try:
        main()
    except KeyboardInterrupt:
        print("\n🛑 Stopped.")
    except Exception as exc:
        print(f"❌ Fatal: {exc}")
